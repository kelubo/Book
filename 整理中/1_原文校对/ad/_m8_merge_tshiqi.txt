\begin{itemize}
\item \textbf{青春期}：
  \begin{itemize}
  \item 开始佩戴胸罩：根据乳房发育情况，选择合适的少女胸罩
  \item 材质选择：优先选择透气、舒适的棉质材质
  \item 款式选择：选择无钢圈、舒适的款式，避免影响乳房发育
  \item 运动时：选择运动胸罩，减少乳房晃动
  \end{itemize}
\item \textbf{孕期}：
  \begin{itemize}
  \item 孕妇胸罩：选择宽松、舒适的哺乳胸罩，提供适当的支撑
  \item 孕妇内裤：选择高腰设计，提供腹部支撑，避免压迫胎儿
  \item 材质选择：优先选择透气、吸汗的棉质材质
  \item 及时更换：随乳房增大及时更换合适尺寸的胸罩
  \end{itemize}
\item \textbf{哺乳期}：
  \begin{itemize}
  \item 哺乳胸罩：方便哺乳，保护乳房和乳头
  \item 防溢乳垫：搭配使用，保持清洁和干燥
  \item 避免过紧：选择宽松、舒适的款式，避免影响乳汁分泌
  \item 支撑性：选择有支撑性的胸罩，减少乳房下垂的风险
  \end{itemize}
\item \textbf{更年期}：
  \begin{itemize}
  \item 材质选择：优先选择透气、吸汗的材质，缓解潮热症状
  \item 支撑性：选择合适的支撑，缓解乳房下垂
  \item 舒适为主：避免过紧或刺激性材质，提高生活质量
  \item 款式与尺寸：选择无钢圈舒适款式，并随乳房变化及时调整尺寸
  \end{itemize}
\end{itemize}
